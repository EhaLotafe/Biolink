const trackView = async () => {
  const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent);
  
  // Récupérer le Referrer (ex: t.co pour Twitter, l.facebook.com, etc.)
  const ref = document.referrer ? new URL(document.referrer).hostname : 'Direct';

  // Récupérer la ville (via une API gratuite comme ipapi.co)
  let city = 'Kinshasa'; // Défaut
  try {
    const res = await fetch('https://ipapi.co/json/');
    const geo = await res.json();
    city = geo.city || 'RDC';
  } catch (e) {}

  await supabase.from('analytics').insert({
    user_id: user.id,
    event_type: 'view',
    device: isMobile ? 'mobile' : 'desktop',
    country: 'RDC',
    city: city,
    referrer: ref
  });
};
